export const AS_SYMPTOMS = [
  '물이 새요', '전기가 안 들어와요', '문이 잘 안 닫혀요',
  '소음·진동이 나요', '깨지거나 떨어졌어요',
];

export function getInitialAsCategory(value: string | string[] | undefined) {
  const category = Array.isArray(value) ? value[0] : value;
  return category && AS_CATEGORIES.includes(category) ? category : AS_CATEGORIES[0];
}

// 선택한 증상도 실제 문의 본문에 저장한다. 사진만 접수한 경우 문제 내용을 추측하지 않는다.
export function buildAsInquiryText(input: {
  title: string;
  description: string;
  categories: string[];
  symptoms: string[];
  attachmentCount: number;
}) {
  const symptoms = [...new Set(input.symptoms.filter((value) => AS_SYMPTOMS.includes(value)))];
  const details = input.description.trim();
  const description = [symptoms.length ? `증상: ${symptoms.join(', ')}` : '', details]
    .filter(Boolean).join('\n\n') || (input.attachmentCount > 0 ? '사진·영상으로 접수한 AS 문의입니다.' : '');
  return {
    title: input.title.trim() || `${input.categories.join(' · ') || '간단 AS'} 문의`,
    description,
  };
}
