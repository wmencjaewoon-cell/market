import Ionicons from '@expo/vector-icons/Ionicons';
import { router, useFocusEffect, type Href } from 'expo-router';
import { useCallback, useMemo, useRef, useState } from 'react';
import { ActivityIndicator, Image, RefreshControl, ScrollView, StyleSheet, Text, TouchableOpacity, View } from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { useAuth } from '../contexts/AuthContext';
import { type AppPalette } from '../contexts/theme';
import { useAppTheme } from '../hooks/use-app-theme';
import { SIMPLE_AS_SERVICE_NAME } from '../lib/appMode';
import { AS_HOME_CATEGORIES } from '../lib/asInquiry';
import { getUnreadNotificationCount } from '../lib/notificationsData';
import { fetchMyRegions, fetchMyRegionSettings } from '../lib/region';
import { supabase } from '../lib/supabase';
import { useTabRefresh } from '../lib/tabRefresh';

type RecentInquiry = { id: number; title: string; created_at: string };
const shortcuts = [
  { label: '내 문의', icon: 'document-text-outline', route: '/my/inquiries' },
  { label: '근처 가게', icon: 'storefront-outline', route: '/store' },
  { label: '장터 보기', icon: 'bag-handle-outline', route: '/explore' },
] as const;

export default function AsHomeScreen() {
  const { user } = useAuth();
  const userId = user?.id;
  const theme = useAppTheme();
  const styles = useMemo(() => createStyles(theme), [theme]);
  const [regionName, setRegionName] = useState('');
  const [unread, setUnread] = useState(0);
  const [recent, setRecent] = useState<{ userId: string; inquiry: RecentInquiry | null } | null>(null);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [failed, setFailed] = useState(false);
  const requestVersion = useRef(0);
  const navigating = useRef(false);
  const currentInquiry = recent && recent.userId === userId ? recent.inquiry : null;

  const navigate = (href: Href) => {
    if (navigating.current) return;
    navigating.current = true;
    router.push(href);
  };

  const loadHome = useCallback(async (refresh = false) => {
    const version = ++requestVersion.current;
    setRefreshing(refresh);
    setLoading(true);
    setFailed(false);
    // 실패한 부가 정보 때문에 문의 버튼을 막지 않는다. 계정 전환 후 늦은 응답도 무시한다.
    const results = await Promise.allSettled([
      getUnreadNotificationCount(),
      Promise.all([fetchMyRegions(), fetchMyRegionSettings()]),
      userId
        ? supabase.from('estimate_requests').select('id, title, created_at')
          .eq('user_id', userId).order('created_at', { ascending: false }).limit(1).maybeSingle()
        : Promise.resolve(null),
    ]);
    if (version !== requestVersion.current) return;
    const [notifications, regions, inquiry] = results;
    setUnread(notifications.status === 'fulfilled' ? notifications.value : 0);
    if (regions.status === 'fulfilled') {
      const [items, settings] = regions.value;
      const active = items.find((item) => item.id === settings?.active_region_id) || items[0];
      setRegionName(active?.region_name || '');
    } else {
      setRegionName('');
    }
    if (inquiry.status === 'fulfilled' && !inquiry.value?.error) {
      setRecent(userId ? { userId, inquiry: inquiry.value?.data ?? null } : null);
    } else {
      setRecent(null);
      setFailed(true);
    }
    setLoading(false);
    setRefreshing(false);
  }, [userId]);

  useFocusEffect(useCallback(() => {
    navigating.current = false;
    setRecent(null);
    setRegionName('');
    setUnread(0);
    void loadHome();
    return () => { requestVersion.current += 1; };
  }, [loadHome]));
  useTabRefresh('home', () => { void loadHome(true); });

  return (
    <SafeAreaView style={styles.screen} edges={['top']}>
      <ScrollView
        contentContainerStyle={styles.content}
        refreshControl={<RefreshControl refreshing={refreshing} onRefresh={() => void loadHome(true)} tintColor={theme.primary} colors={['#166534']} />}
      >
        <View style={styles.header}>
          <Image source={require('../assets/images/interior-market.jpg')} style={styles.logo} accessibilityIgnoresInvertColors />
          <Text style={styles.brand}>{SIMPLE_AS_SERVICE_NAME}</Text>
          <TouchableOpacity style={styles.iconButton} accessibilityRole="button" accessibilityLabel={`알림${unread ? `, 읽지 않은 알림 ${unread}개` : ''}`} onPress={() => navigate('/my/notifications' as Href)}>
            <Ionicons name="notifications-outline" size={24} color={theme.text} />
            {unread > 0 ? <View style={styles.badge}><Text style={styles.badgeText}>{unread > 99 ? '99+' : unread}</Text></View> : null}
          </TouchableOpacity>
        </View>
        <TouchableOpacity style={styles.location} accessibilityRole="button" accessibilityLabel="내 동네 설정" onPress={() => navigate('/(tabs)/home/regions' as Href)}>
          <Ionicons name="location-outline" size={15} color={theme.textMuted} />
          <Text style={styles.locationText} numberOfLines={1}>{regionName || '내 동네 설정'}</Text>
          <Ionicons name="chevron-down" size={14} color={theme.textMuted} />
        </TouchableOpacity>

        <View style={styles.intro}>
          <Text style={styles.eyebrow}>사진·영상으로 간편하게</Text>
          <Text style={styles.title}>집수리·AS 문의</Text>
          <Text style={styles.subtitle}>고장 난 곳을 보여주세요.{`\n`}확인 후 상담과 업체 연결을 도와드려요.</Text>
          <TouchableOpacity style={styles.primaryButton} activeOpacity={0.85} accessibilityRole="button" onPress={() => navigate('/estimate/create' as Href)}>
            <Ionicons name="camera-outline" size={24} color="#fff" />
            <Text style={styles.primaryText}>수리·AS 문의하기</Text>
            <Ionicons name="arrow-forward" size={20} color="#fff" />
          </TouchableOpacity>
        </View>

        <View style={styles.shortcuts}>
          {shortcuts.map((item) => (
            <TouchableOpacity key={item.label} style={styles.shortcut} accessibilityRole="button" accessibilityLabel={item.label} onPress={() => navigate(item.route as Href)}>
              <Ionicons name={item.icon} size={25} color={theme.text} />
              <Text style={styles.shortcutText}>{item.label}</Text>
            </TouchableOpacity>
          ))}
        </View>

        <View style={styles.section}>
          <Text style={styles.sectionTitle}>어디를 고쳐야 하나요?</Text>
          <View style={styles.categoryGrid}>
            {AS_HOME_CATEGORIES.map((item) => (
              <View key={item.category} style={styles.categoryCell}>
                <TouchableOpacity style={styles.category} accessibilityRole="button" accessibilityLabel={`${item.label} 문의`} onPress={() => navigate({ pathname: '/estimate/create', params: { category: item.category } } as Href)}>
                  <Ionicons name={item.icon} size={26} color={theme.scheme === 'dark' ? theme.text : item.color} />
                  <Text style={styles.categoryTitle}>{item.label}</Text>
                  <Text style={styles.categoryDetail}>{item.detail}</Text>
                </TouchableOpacity>
              </View>
            ))}
          </View>
        </View>

        <View style={styles.section}>
          <View style={styles.sectionHeader}>
            <Text style={styles.sectionTitle}>최근 문의</Text>
            <TouchableOpacity style={styles.textButton} accessibilityRole="button" onPress={() => navigate('/my/inquiries' as Href)}>
              <Text style={styles.linkText}>전체 보기</Text><Ionicons name="chevron-forward" size={16} color={theme.textMuted} />
            </TouchableOpacity>
          </View>
          {!userId ? (
            <TouchableOpacity style={styles.recentRow} accessibilityRole="button" onPress={() => navigate('/login?redirect=/my/inquiries' as Href)}>
              <Ionicons name="person-circle-outline" size={28} color={theme.textMuted} />
              <Text style={styles.rowText}>로그인하고 내 문의를 확인하세요</Text>
              <Ionicons name="chevron-forward" size={18} color={theme.textMuted} />
            </TouchableOpacity>
          ) : loading ? (
            <View style={styles.recentRow}><ActivityIndicator color={theme.primary} /><Text style={styles.muted}>문의 내역을 불러오는 중...</Text></View>
          ) : failed ? (
            <TouchableOpacity style={styles.recentRow} accessibilityRole="button" accessibilityLabel="문의 내역 다시 불러오기" onPress={() => void loadHome(true)}>
              <Text style={styles.rowText}>문의 내역을 불러오지 못했어요</Text><Ionicons name="refresh-outline" size={22} color={theme.text} />
            </TouchableOpacity>
          ) : currentInquiry ? (
            <TouchableOpacity style={styles.recentRow} accessibilityRole="button" onPress={() => navigate('/my/inquiries' as Href)}>
              <Ionicons name="document-text-outline" size={25} color={theme.primary} />
              <View style={styles.recentText}><Text style={styles.rowText} numberOfLines={2}>{currentInquiry.title}</Text><Text style={styles.muted}>{new Date(currentInquiry.created_at).toLocaleDateString('ko-KR')} 접수</Text></View>
              <Ionicons name="chevron-forward" size={18} color={theme.textMuted} />
            </TouchableOpacity>
          ) : <View style={styles.recentRow}><Ionicons name="document-outline" size={24} color={theme.textMuted} /><Text style={styles.muted}>아직 접수한 문의가 없어요</Text></View>}
        </View>

        <TouchableOpacity style={styles.marketRow} accessibilityRole="button" onPress={() => navigate('/explore' as Href)}>
          <Ionicons name="bag-handle-outline" size={25} color={theme.text} />
          <View style={styles.recentText}><Text style={styles.rowText}>필요한 물품도 찾아보세요</Text><Text style={styles.muted}>우리 동네 거래 · 나눔 · 구함</Text></View>
          <Ionicons name="arrow-forward" size={20} color={theme.text} />
        </TouchableOpacity>
      </ScrollView>
    </SafeAreaView>
  );
}

function createStyles(theme: AppPalette) {
  return StyleSheet.create({
    screen: { flex: 1, backgroundColor: theme.background },
    content: { width: '100%', maxWidth: 640, alignSelf: 'center', padding: 20, paddingBottom: 32 },
    header: { flexDirection: 'row', alignItems: 'center', gap: 10 },
    logo: { width: 36, height: 36, borderRadius: 8 },
    brand: { flex: 1, color: theme.text, fontSize: 22, fontWeight: '800' },
    iconButton: { width: 44, height: 44, alignItems: 'center', justifyContent: 'center' },
    badge: { position: 'absolute', top: 0, right: 0, minWidth: 18, borderRadius: 10, paddingHorizontal: 4, backgroundColor: '#166534' },
    badgeText: { color: '#fff', fontSize: 11, lineHeight: 18, textAlign: 'center', fontWeight: '700' },
    location: { minHeight: 44, flexDirection: 'row', alignItems: 'center', alignSelf: 'flex-start', gap: 4, maxWidth: '100%' },
    locationText: { color: theme.textMuted, fontSize: 13, flexShrink: 1 },
    intro: { paddingTop: 18, paddingBottom: 24, gap: 10 },
    eyebrow: { color: theme.primary, fontSize: 14, fontWeight: '700' },
    title: { color: theme.text, fontSize: 28, lineHeight: 38, fontWeight: '800', letterSpacing: 0 },
    subtitle: { color: theme.textMuted, fontSize: 15, lineHeight: 24 },
    primaryButton: { marginTop: 12, minHeight: 60, flexDirection: 'row', alignItems: 'center', padding: 16, gap: 10, borderRadius: 8, backgroundColor: '#166534' },
    primaryText: { flex: 1, color: '#fff', fontSize: 17, lineHeight: 24, fontWeight: '800' },
    shortcuts: { flexDirection: 'row', gap: 8, paddingBottom: 24 },
    shortcut: { flex: 1, minWidth: 0, minHeight: 88, paddingVertical: 14, paddingHorizontal: 4, borderRadius: 8, backgroundColor: theme.surfaceSoft, alignItems: 'center', justifyContent: 'center', gap: 8 },
    shortcutText: { fontSize: 13, lineHeight: 20, fontWeight: '700', color: theme.text, textAlign: 'center' },
    section: { borderTopWidth: 1, borderTopColor: theme.border, paddingTop: 24, paddingBottom: 24, gap: 16 },
    sectionTitle: { color: theme.text, fontSize: 18, fontWeight: '800', flexShrink: 1 },
    sectionHeader: { flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between', gap: 8 },
    categoryGrid: { flexDirection: 'row', flexWrap: 'wrap', marginHorizontal: -4, rowGap: 8 },
    categoryCell: { width: '33.333333%', paddingHorizontal: 4 },
    category: { minHeight: 118, height: '100%', paddingVertical: 14, paddingHorizontal: 3, gap: 7, borderRadius: 8, borderWidth: 1, borderColor: theme.border, alignItems: 'center', justifyContent: 'center' },
    categoryTitle: { color: theme.text, fontSize: 14, lineHeight: 20, fontWeight: '700', textAlign: 'center' },
    categoryDetail: { color: theme.textMuted, fontSize: 11, lineHeight: 17, textAlign: 'center' },
    textButton: { minHeight: 44, flexDirection: 'row', alignItems: 'center', gap: 2 },
    linkText: { color: theme.textMuted, fontSize: 13 },
    recentRow: { minHeight: 68, flexDirection: 'row', alignItems: 'center', gap: 12, paddingVertical: 8 },
    recentText: { flex: 1, minWidth: 0, gap: 6 },
    rowText: { color: theme.text, fontSize: 15, lineHeight: 23, fontWeight: '700', flexShrink: 1 },
    muted: { color: theme.textMuted, fontSize: 13, lineHeight: 20, flexShrink: 1 },
    marketRow: { borderTopWidth: 1, borderTopColor: theme.border, paddingTop: 24, paddingBottom: 8, flexDirection: 'row', alignItems: 'center', gap: 12 },
  });
}
